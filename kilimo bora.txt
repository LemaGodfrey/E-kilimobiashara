<script>
  // ---------- PRODUCTS DATABASE (farm goods) ----------
  const PRODUCTS = [
    { id: "maize", name: "Maize (dry)", basePrice: 60, unit: "per kg", img: "https://picsum.photos/id/144/300/180" },
    { id: "beans", name: "Mixed Beans", basePrice: 120, unit: "per kg", img: "https://picsum.photos/id/159/300/180" },
    { id: "potatoes", name: "Irish Potatoes", basePrice: 80, unit: "per kg", img: "https://picsum.photos/id/157/300/180" },
    { id: "tomatoes", name: "Fresh Tomatoes", basePrice: 100, unit: "per kg", img: "https://picsum.photos/id/110/300/180" },
    { id: "onions", name: "Red Onions", basePrice: 90, unit: "per kg", img: "https://picsum.photos/id/122/300/180" },
    { id: "kale", name: "Sukuma Wiki", basePrice: 40, unit: "per bunch", img: "https://picsum.photos/id/104/300/180" },
    { id: "milk", name: "Fresh Milk", basePrice: 50, unit: "per litre", img: "https://picsum.photos/id/130/300/180" },
    { id: "eggs", name: "Free-Range Eggs", basePrice: 15, unit: "per piece", img: "https://picsum.photos/id/102/300/180" }
  ];

  // Multipliers based on selection
  const LOCATION_MULT = 
  {
    "Nairobi": 1.25,
    "Kiambu": 1.0,
    "Nakuru": 0.92,
    "Kisumu": 0.95,
    "Mombasa": 1.18,
    "Eldoret": 0.88
  };
  const DEMAND_MULT = 
  {
    "Low": 0.85,
    "Medium": 1.0,
    "High": 1.30
  };
  const SUPPLIER_MULT = 
  {
    "Small-scale farmer": 1.0,
    "Large-scale farmer": 0.92   // economies of scale, cheaper
  };
  const QUALITY_MULT = 
  {
    "Standard": 1.0,
    "Premium": 1.28
  };

  // Helper: compute final price per unit
  function computePrice(base, location, demand, supplier, quality) 
  {
    let mult = LOCATION_MULT[location] * DEMAND_MULT[demand] * SUPPLIER_MULT[supplier] * QUALITY_MULT[quality];
    return +(base * mult).toFixed(2);
  }

  // Generate stock / availability message based on demand & supplier
  function getAvailabilityNote(demand, supplier) 
  {
    if (demand === "High" && supplier === "Small-scale farmer") return "⚠️ Limited stock! Order soon";
    if (demand === "High") return "🔥 High demand - few left";
    if (demand === "Low") return "✅ Good availability, best prices";
    return "✔️ In stock";
  }

  // Cart structure: each item stores { id, name, unit, quantity, selectedOptions, pricePerUnit, totalPrice, img }
  let cart = [];  // array of objects

  // Load cart from localStorage
  function loadCart() 
  {
    const stored = localStorage.getItem("kilimoBoraCart");
    if (stored) {
      try {
        let parsed = JSON.parse(stored);
        if (Array.isArray(parsed)) 
        {
          cart = parsed;
        }
      } catch(e) {}
    }
    renderCartUI();
    updateCartBadge();
  }

  function saveCart() 
  {
    localStorage.setItem("kilimoBoraCart", JSON.stringify(cart));
  }

  function updateCartBadge() 
  {
    const totalItems = cart.reduce((sum, item) => sum + item.quantity, 0);
    document.getElementById("cartItemCount").innerText = totalItems;
  }

  // Render product cards with dynamic selectors
  function renderProducts() 
  {
    const grid = document.getElementById("productGrid");
    if (!grid) return;
    let html = "";
    PRODUCTS.forEach(prod => 
    {
      // default selections
      const defaultLocation = "Kiambu";
      const defaultDemand = "Medium";
      const defaultSupplier = "Small-scale farmer";
      const defaultQuality = "Standard";
      const defaultPrice = computePrice(prod.basePrice, defaultLocation, defaultDemand, defaultSupplier, defaultQuality);
      html += `
        <div class="product-card" data-prod-id="${prod.id}">
          <img class="product-img" src="${prod.img}" alt="${prod.name}" loading="lazy" onerror="this.src='https://picsum.photos/id/180/300/180'">
          <div class="product-info">
            <div class="product-title">${prod.name}</div>
            <div class="unit">${prod.unit}</div>
            
            <div class="options-group">
              <select class="opt-select location-select" data-id="${prod.id}">
                ${Object.keys(LOCATION_MULT).map(loc => `<option value="${loc}" ${loc === defaultLocation ? 'selected' : ''}>📍 ${loc}</option>`).join('')}
              </select>
              <select class="opt-select demand-select" data-id="${prod.id}">
                ${Object.keys(DEMAND_MULT).map(d => `<option value="${d}" ${d === defaultDemand ? 'selected' : ''}>📈 Demand: ${d}</option>`).join('')}
              </select>
            </div>
            <div class="options-group">
              <select class="opt-select supplier-select" data-id="${prod.id}">
                ${Object.keys(SUPPLIER_MULT).map(s => `<option value="${s}" ${s === defaultSupplier ? 'selected' : ''}>👩‍🌾 ${s}</option>`).join('')}
              </select>
              <select class="opt-select quality-select" data-id="${prod.id}">
                ${Object.keys(QUALITY_MULT).map(q => `<option value="${q}" ${q === defaultQuality ? 'selected' : ''}>⭐ ${q}</option>`).join('')}
              </select>
            </div>
            
            <div class="price-preview">
              <span class="final-price" id="priceDisplay-${prod.id}">KES ${defaultPrice}</span>
              <div class="price-note">per ${prod.unit}</div>
            </div>
            <div class="stock-badge" id="stockMsg-${prod.id}">${getAvailabilityNote(defaultDemand, defaultSupplier)}</div>
            
            <div class="quantity-row">
              <span>Quantity:</span>
              <div class="qty-selector">
                <button class="qty-btn decr-qty" data-id="${prod.id}">−</button>
                <span id="qtyVal-${prod.id}">1</span>
                <button class="qty-btn incr-qty" data-id="${prod.id}">+</button>
              </div>
            </div>
            <button class="add-to-cart-btn" data-id="${prod.id}"><i class="fas fa-cart-plus"></i> Add to basket</button>
          </div>
        </div>
      `;
    });
    grid.innerHTML = html;

    // Attach dynamic price update listeners for each product
    PRODUCTS.forEach(prod => {
      const locSelect = document.querySelector(`.location-select[data-id="${prod.id}"]`);
      const demandSelect = document.querySelector(`.demand-select[data-id="${prod.id}"]`);
      const supplierSelect = document.querySelector(`.supplier-select[data-id="${prod.id}"]`);
      const qualitySelect = document.querySelector(`.quality-select[data-id="${prod.id}"]`);
      const priceSpan = document.getElementById(`priceDisplay-${prod.id}`);
      const stockSpan = document.getElementById(`stockMsg-${prod.id}`);
      
      function updatePriceAndStock() 
      {
        const location = locSelect.value;
        const demand = demandSelect.value;
        const supplier = supplierSelect.value;
        const quality = qualitySelect.value;
        const newPrice = computePrice(prod.basePrice, location, demand, supplier, quality);
        if (priceSpan) priceSpan.innerText = `KES ${newPrice}`;
        if (stockSpan) stockSpan.innerText = getAvailabilityNote(demand, supplier);
      }
      locSelect.addEventListener('change', updatePriceAndStock);
      demandSelect.addEventListener('change', updatePriceAndStock);
      supplierSelect.addEventListener('change', updatePriceAndStock);
      qualitySelect.addEventListener('change', updatePriceAndStock);
      
      // quantity buttons
      const qtySpan = document.getElementById(`qtyVal-${prod.id}`);
      const decr = document.querySelector(`.decr-qty[data-id="${prod.id}"]`);
      const incr = document.querySelector(`.incr-qty[data-id="${prod.id}"]`);
      let qty = 1;
      if (decr && incr && qtySpan) 
      {
        decr.addEventListener('click', () => 
        {
          qty = Math.max(1, qty - 1);
          qtySpan.innerText = qty;
        });
        incr.addEventListener('click', () => 
        {
          qty = qty + 1;
          qtySpan.innerText = qty;
        });
      }
      // attach add-to-cart with current selections & quantity
      const addBtn = document.querySelector(`.add-to-cart-btn[data-id="${prod.id}"]`);
      if (addBtn) 
      {
        addBtn.addEventListener('click', () => 
        {
          const location = locSelect.value;
          const demand = demandSelect.value;
          const supplier = supplierSelect.value;
          const quality = qualitySelect.value;
          let quantity = parseInt(document.getElementById(`qtyVal-${prod.id}`).innerText);
          if (isNaN(quantity) || quantity < 1) quantity = 1;
          const pricePerUnit = computePrice(prod.basePrice, location, demand, supplier, quality);
          const totalItemPrice = +(pricePerUnit * quantity).toFixed(2);
          const selectedOptions = { location, demand, supplier, quality };
          
          // create cart item (we'll push, but combine identical options? For simplicity allow duplicates per configuration)
          // We can combine if same product and same options to keep cart clean.
          const existingIndex = cart.findIndex(item => 
            item.id === prod.id && 
            item.selectedOptions.location === location &&
            item.selectedOptions.demand === demand &&
            item.selectedOptions.supplier === supplier &&
            item.selectedOptions.quality === quality
          );
          if (existingIndex !== -1) 
          {
            cart[existingIndex].quantity += quantity;
            cart[existingIndex].totalPrice = +(cart[existingIndex].pricePerUnit * cart[existingIndex].quantity).toFixed(2);
          } else 
          {
            cart.push(
              {
              id: prod.id,
              name: prod.name,
              unit: prod.unit,
              quantity: quantity,
              selectedOptions: selectedOptions,
              pricePerUnit: pricePerUnit,
              totalPrice: totalItemPrice,
              img: prod.img
            });
          }
          saveCart();
          renderCartUI();
          updateCartBadge();
          // provide subtle feedback
          addBtn.style.transform = "scale(0.96)";
          setTimeout(() => { addBtn.style.transform = ""; }, 150);
        });
      }
    });
  }

  // Render cart UI
  function renderCartUI() 
  {
    const cartContainer = document.getElementById("cartItemsList");
    const totalSpan = document.getElementById("cartTotal");
    if (!cartContainer) return;
    
    if (cart.length === 0) 
    {
      cartContainer.innerHTML = `<div class="cart-empty">🌾 Your basket is empty. Support local farmers!</div>`;
      if (totalSpan) totalSpan.innerText = "KES 0";
      return;
    }
    
    let cartHtml = "";
    let overallTotal = 0;
    cart.forEach((item, idx) => 
    {
      const opt = item.selectedOptions;
      const optionText = `${opt.location} | ${opt.demand} demand | ${opt.supplier} | ${opt.quality}`;
      overallTotal += item.totalPrice;
      cartHtml += `
        <div class="cart-item" data-cart-index="${idx}">
          <img class="cart-item-img" src="${item.img}" onerror="this.src='https://picsum.photos/id/129/55/55'">
          <div class="cart-item-details">
            <div class="cart-item-name">${item.name} (${item.unit})</div>
            <div class="cart-item-options">${optionText}</div>
            <div class="cart-item-price">KES ${item.pricePerUnit} / ${item.unit.split(' ')[0]}</div>
            <div class="cart-item-actions">
              <button class="cart-qty-btn cart-decr" data-idx="${idx}">−</button>
              <span style="font-weight:600; margin:0 4px;">${item.quantity}</span>
              <button class="cart-qty-btn cart-incr" data-idx="${idx}">+</button>
              <button class="cart-remove" data-idx="${idx}"><i class="fas fa-trash-alt"></i></button>
            </div>
          </div>
          <div style="font-weight:700;">KES ${item.totalPrice}</div>
        </div>
      `;
    });
    cartContainer.innerHTML = cartHtml;
    if (totalSpan) totalSpan.innerText = `KES ${overallTotal.toFixed(2)}`;
    
    // Attach cart event handlers
    document.querySelectorAll('.cart-decr').forEach(btn => 
    {
      btn.addEventListener('click', (e) => 
      {
        const idx = parseInt(btn.dataset.idx);
        if (!isNaN(idx) && cart[idx]) 
        {
          if (cart[idx].quantity > 1) 
          {
            cart[idx].quantity--;
            cart[idx].totalPrice = +(cart[idx].pricePerUnit * cart[idx].quantity).toFixed(2);
          } else 
          {
            cart.splice(idx, 1);
          }
          saveCart();
          renderCartUI();
          updateCartBadge();
        }
      });
    });
    document.querySelectorAll('.cart-incr').forEach(btn => 
    {
      btn.addEventListener('click', () => 
      {
        const idx = parseInt(btn.dataset.idx);
        if (!isNaN(idx) && cart[idx]) 
        {
          cart[idx].quantity++;
          cart[idx].totalPrice = +(cart[idx].pricePerUnit * cart[idx].quantity).toFixed(2);
          saveCart();
          renderCartUI();
          updateCartBadge();
        }
      });
    });
    document.querySelectorAll('.cart-remove').forEach(btn => 
    {
      btn.addEventListener('click', () => 
      {
        const idx = parseInt(btn.dataset.idx);
        if (!isNaN(idx)) {
          cart.splice(idx, 1);
          saveCart();
          renderCartUI();
          updateCartBadge();
        }
      });
    });
  }
  
  // Clear cart handler
  function setupGlobalActions() 
  {
    const clearBtn = document.getElementById("clearCartBtn");
    if (clearBtn) 
    {
      clearBtn.addEventListener("click", () => 
      {
        if (cart.length && confirm("Remove all items from basket?")) 
        {
          cart = [];
          saveCart();
          renderCartUI();
          updateCartBadge();
        }
      });
    }
    const checkoutBtn = document.getElementById("checkoutBtn");
    if (checkoutBtn) {
      checkoutBtn.addEventListener("click", () => 
      {
        if (cart.length === 0) 
        {
          alert("Your basket is empty. Please add fresh produce.");
          return;
        }
        let total = cart.reduce((sum, i) => sum + i.totalPrice, 0);
        let itemSummary = cart.map(i => `${i.name} x${i.quantity} (${i.selectedOptions.location})`).join(", ");
        alert(`✅ Order received! 🌍 Thank you for supporting Kenyan farmers.\n\nItems: ${itemSummary}\nTotal: KES ${total.toFixed(2)}\n\nYour produce will be delivered based on location pricing. Asante sana!`);
        // Optional: keep cart or reset? Usually keep for demo but we can clear if needed. Better to let user decide.
        // For better experience, we can ask: "Clear cart after order?" but we'll just keep for flexibility.
        // But many e-com sites keep cart until user clears. We'll not auto-clear.
      });
    }
  }
  
  // Initialization
  function init() 
  {
    loadCart();        // loads cart from storage
    renderProducts();  // build dynamic product cards
    setupGlobalActions();
  }
  
  init();
</script>
